tests/test_simulation.py

from gedt.simulation.engine import GEDTEngine
from gedt.core.config import SimulationConfig


def test_simulation_runs():

    config = SimulationConfig(
        population=100,
        years=10,
        innovation_rate=0.05,
        seed=42
    )


    engine = GEDTEngine()

    result = engine.run(
        config
    )


    assert result is not None


def test_output_range():

    config = SimulationConfig(
        population=100,
        years=10
    )


    result = GEDTEngine().run(
        config
    )


    assert (
        0 <= result["final_productivity"] <= 1
    )

from gedt.statistics.metrics import Statistics


def test_mean_calculation():

    stats = Statistics()

    values=[1,2,3,4]

    result=stats.mean(values)

    assert result == 2.5
