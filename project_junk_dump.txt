import numpy as np
class Population:
    def __init__(self, size):
        self.skill = np.random.random(size)
        self.resources = np.random.random(size)
        self.adaptability = np.random.random(size)
    def innovate(self, rate):
        self.skill += rate * self.adaptability
        self.skill = np.clip(
            self.skill,
            0,
            1
        )
    def productivity(self):
        return np.mean(
            self.skill * self.resources
        )
HISTORICAL VECTORISED POPULATION
--------------------------------
import numpy as np
class Population:
    def __init__(self, size, seed=42):
        rng = np.random.default_rng(seed)
        self.skill = rng.random(size)
        self.resources = rng.random(size)
        self.adaptability = rng.random(size)
    def update(self, innovation_rate):
        self.skill += (
            innovation_rate *
            self.adaptability
        )
        np.clip(
            self.skill,
            0,
            1,
            out=self.skill
        )
    def productivity(self):
        return float(
            np.mean(
                self.skill *
                self.resources
            )
        )
HISTORICAL SIMULATION ENGINE

from .population import Population
class SimulationEngine:
    def run(
        self,
        population_size,
        years,
        innovation_rate,
        seed=42
    ):
        population = Population(
            population_size,
            seed
        )
        history = []
        for year in range(years):
            population.update(
                innovation_rate
            )
            history.append(
                population.productivity()
            )
        return {
            "final_productivity": history[-1],
            "history": history
        }
